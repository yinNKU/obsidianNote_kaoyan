## 图的应用算法代码实现

图的核心应用算法（MST、最短路径）的 C++ 完整实现，可直接编译运行。

### Prim 算法（邻接矩阵）

```cpp
#include <iostream>
#include <vector>

using namespace std;

const int N = 105;
const int INF = 0x3f3f3f3f;

int n,m;

int g[N][N];

int lowcost[N];
int parent[N];
bool vis[N];

int prim(int start)
{
    for(int i=1;i<=n;i++)
    {
        lowcost[i]=INF;
        parent[i]=-1;
        vis[i]=false;
    }

    lowcost[start]=0;

    int mstWeight=0;

    for(int i=1;i<=n;i++)
    {
        int u=-1;

        for(int j=1;j<=n;j++)
        {
            if(!vis[j] &&
               (u==-1 || lowcost[j]<lowcost[u]))
            {
                u=j;
            }
        }

        if(u==-1)
            return -1;      //图不连通

        vis[u]=true;

        mstWeight+=lowcost[u];

        for(int v=1;v<=n;v++)
        {
            if(!vis[v] &&
               g[u][v] < lowcost[v])
            {
                lowcost[v]=g[u][v];

                parent[v]=u;
            }
        }
    }

    return mstWeight;
}

int main()
{
    cin>>n>>m;

    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            if(i==j)
                g[i][j]=0;
            else
                g[i][j]=INF;
        }
    }

    for(int i=0;i<m;i++)
    {
        int u,v,w;

        cin>>u>>v>>w;

        g[u][v]=g[v][u]=w;
    }

    int ans=prim(1);

    cout<<"MST权值="<<ans<<"\n\n";

    cout<<"生成树边：\n";

    for(int i=1;i<=n;i++)
    {
        if(parent[i]!=-1)
        {
            cout
                <<parent[i]
                <<" - "
                <<i
                <<" ("
                <<g[parent[i]][i]
                <<")\n";
        }
    }
}
```

### Kruskal 算法（并查集 + 边集数组）

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int N=100005;

struct Edge
{
    int u;
    int v;
    int w;

    bool operator<(const Edge& other) const
    {
        return w<other.w;
    }
};

vector<Edge> edges;

int fa[N];

int find(int x)
{
    if(fa[x]==x)
        return x;

    return fa[x]=find(fa[x]);
}

int kruskal(int n)
{
    for(int i=1;i<=n;i++)
        fa[i]=i;

    sort(edges.begin(),edges.end());

    int mstWeight=0;

    int cnt=0;

    vector<Edge> mstEdges;

    for(auto &e:edges)
    {
        int fu=find(e.u);
        int fv=find(e.v);

        if(fu!=fv)
        {
            fa[fu]=fv;

            mstWeight+=e.w;

            cnt++;

            mstEdges.push_back(e);

            if(cnt==n-1)
                break;
        }
    }

    if(cnt!=n-1)
    {
        cout<<"图不连通\n";
        return -1;
    }

    cout<<"生成树边：\n";

    for(auto &e:mstEdges)
    {
        cout
            <<e.u
            <<" - "
            <<e.v
            <<" ("
            <<e.w
            <<")\n";
    }

    return mstWeight;
}

int main()
{
    int n,m;

    cin>>n>>m;

    for(int i=0;i<m;i++)
    {
        Edge e;

        cin>>e.u>>e.v>>e.w;

        edges.push_back(e);
    }

    int ans=kruskal(n);

    cout<<"\nMST权值="
        <<ans
        <<'\n';
}
```

### Dijkstra 算法（邻接矩阵）

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int N = 105;
const int INF = 0x3f3f3f3f;

int n, m;

int g[N][N];      // 邻接矩阵
int dist[N];      // 最短距离
int path[N];      // 前驱结点
bool vis[N];

void dijkstra(int s)
{
    for(int i = 1; i <= n; i++)
    {
        dist[i] = INF;
        path[i] = -1;
        vis[i] = false;
    }

    dist[s] = 0;

    for(int i = 1; i <= n; i++)
    {
        int u = -1;

        // 找当前未访问且dist最小的点
        for(int j = 1; j <= n; j++)
        {
            if(!vis[j] &&
               (u == -1 || dist[j] < dist[u]))
            {
                u = j;
            }
        }

        if(u == -1)
            break;

        vis[u] = true;

        // 松弛
        for(int v = 1; v <= n; v++)
        {
            if(!vis[v] &&
               g[u][v] != INF &&
               dist[v] > dist[u] + g[u][v])
            {
                dist[v] = dist[u] + g[u][v];

                path[v] = u;      // 记录前驱
            }
        }
    }
}

// 恢复路径
vector<int> getPath(int s,int t)
{
    vector<int> ans;

    for(int cur=t; cur!=-1; cur=path[cur])
        ans.push_back(cur);

    reverse(ans.begin(),ans.end());

    if(ans.empty() || ans[0]!=s)
        return {};

    return ans;
}

void printPath(int s,int t)
{
    vector<int> p=getPath(s,t);

    if(p.empty())
    {
        cout<<"不存在路径\n";
        return;
    }

    for(int i=0;i<p.size();i++)
    {
        if(i) cout<<" -> ";
        cout<<p[i];
    }

    cout<<'\n';
}

int main()
{
    cin>>n>>m;

    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            if(i==j)
                g[i][j]=0;
            else
                g[i][j]=INF;
        }
    }

    for(int i=0;i<m;i++)
    {
        int u,v,w;
        cin>>u>>v>>w;

        g[u][v]=w;   // 有向图

        // 无向图再加：
        // g[v][u]=w;
    }

    int s=1;

    dijkstra(s);

    for(int i=1;i<=n;i++)
    {
        cout<<"dist["<<i<<"] = "
            <<dist[i]<<'\n';
    }

    cout<<"\n1到"<<n<<"的最短路径:\n";

    printPath(1,n);

    return 0;
}
```

### Dijkstra 算法（优先队列优化）

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

using namespace std;

const int N = 100005;
const int INF = 0x3f3f3f3f;

vector<pair<int,int>> g[N];

int dist[N];
int path[N];
bool vis[N];

int n,m;

void dijkstra(int s)
{
    for(int i=1;i<=n;i++)
    {
        dist[i]=INF;
        path[i]=-1;
        vis[i]=false;
    }

    priority_queue<
        pair<int,int>,
        vector<pair<int,int>>,
        greater<pair<int,int>>
    > pq;

    dist[s]=0;

    pq.push({0,s});

    while(!pq.empty())
    {
        auto [d,u]=pq.top();
        pq.pop();

        if(vis[u])
            continue;

        vis[u]=true;

        for(auto [v,w]:g[u])
        {
            if(dist[v] > dist[u] + w)
            {
                dist[v] = dist[u] + w;

                path[v] = u;

                pq.push({dist[v],v});
            }
        }
    }
}

vector<int> getPath(int s,int t)
{
    vector<int> ans;

    for(int cur=t; cur!=-1; cur=path[cur])
        ans.push_back(cur);

    reverse(ans.begin(),ans.end());

    if(ans.empty() || ans[0]!=s)
        return {};

    return ans;
}

void printPath(int s,int t)
{
    vector<int> p=getPath(s,t);

    if(p.empty())
    {
        cout<<"不存在路径\n";
        return;
    }

    for(int i=0;i<p.size();i++)
    {
        if(i) cout<<" -> ";
        cout<<p[i];
    }

    cout<<'\n';
}

int main()
{
    cin>>n>>m;

    for(int i=0;i<m;i++)
    {
        int u,v,w;

        cin>>u>>v>>w;

        g[u].push_back({v,w});   // 有向图

        // 无向图：
        // g[v].push_back({u,w});
    }

    int s=1;

    dijkstra(s);

    for(int i=1;i<=n;i++)
    {
        cout<<"dist["<<i<<"] = "
            <<dist[i]<<'\n';
    }

    cout<<"\n1到"<<n<<"的最短路径:\n";

    printPath(1,n);

    return 0;
}
```

## Floyd算法
```c
#include <iostream>
using namespace std;

const int N = 105;
const int INF = 0x3f3f3f3f;

int dist[N][N];

int main() {
    int n, m;
    cin >> n >> m;

    // 初始化
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= n; j++) {
            if(i == j)
                dist[i][j] = 0;
            else
                dist[i][j] = INF;
        }
    }

    // 读入边
    for(int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;

        // 重边取最小值
        dist[u][v] = min(dist[u][v], w);
    }

    // Floyd
    for(int k = 1; k <= n; k++)
        for(int i = 1; i <= n; i++)
            for(int j = 1; j <= n; j++)
                if(dist[i][k] != INF && dist[k][j] != INF)
                    dist[i][j] = min(dist[i][j],
                                     dist[i][k] + dist[k][j]);

    // 输出
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= n; j++) {
            if(dist[i][j] == INF)
                cout << "INF ";
            else
                cout << dist[i][j] << ' ';
        }
        cout << '\n';
    }
}
```
